Multi-Backend Storage Demo (Local / Root File / Supabase)

Writes directly to ./gamekit-save.json in your project root folder.
Supabase Table SQL (Run once in Supabase SQL Editor) 
create table if not exists public.gamekit_saves (
  profile_id text primary key,
  data jsonb not null default '{}'::jsonb,
  updated_at timestamptz not null default now()
);
alter table public.gamekit_saves enable row level security;
create policy "Allow public save access" on public.gamekit_saves
  for all using (true) with check (true);
